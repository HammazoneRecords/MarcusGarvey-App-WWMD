-- Solob Wrapper V1 ? Canonical SQLite Schema
-- Purpose: Anchor-referential, audit-first memory spine

PRAGMA foreign_keys = ON;

-- =========================================================
-- TABLE: anchors
-- Indexes external, read-only anchor sources (PDF, JSON, text)
-- =========================================================
CREATE TABLE anchors (
    anchor_id TEXT PRIMARY KEY,
    anchor_type TEXT NOT NULL,          -- lexicon | book | letter | other
    title TEXT NOT NULL,
    source_path TEXT NOT NULL,           -- filesystem path to anchor
    source_format TEXT NOT NULL,         -- json | pdf | txt
    status TEXT NOT NULL,                -- canon | working
    provenance TEXT NOT NULL,            -- human-entered source description
    created_at TEXT NOT NULL             -- ISO-8601 timestamp
);

-- =========================================================
-- TABLE: chunks
-- Atomic, referenceable statements extracted from anchors
-- =========================================================
CREATE TABLE chunks (
    chunk_id TEXT PRIMARY KEY,
    anchor_id TEXT NOT NULL,
    anchor_locator TEXT NOT NULL,        -- page, section, lexicon row, etc.
    content TEXT NOT NULL,               -- exact extracted text (no mutation)
    truth_type TEXT NOT NULL CHECK (
        truth_type IN (
            'definition',
            'event',
            'empirical',
            'interpretive',
            'causal',
            'relational'
        )
    ),
    mutation_mode TEXT NOT NULL CHECK (
        mutation_mode IN (
            'append-only',
            'versioned',
            'time-bounded',
            're-index-only'
        )
    ),
    confidence REAL CHECK (
        confidence IS NULL
        OR (confidence >= 0 AND confidence <= 1)
    ),
    created_at TEXT NOT NULL,

    FOREIGN KEY (anchor_id)
        REFERENCES anchors(anchor_id)
        ON DELETE RESTRICT
);

-- =========================================================
-- TABLE: provenance_notes
-- Non-mutative annotations and corrections
-- =========================================================
CREATE TABLE provenance_notes (
    note_id TEXT PRIMARY KEY,
    target_type TEXT NOT NULL,           -- anchor | chunk
    target_id TEXT NOT NULL,
    note TEXT NOT NULL,
    author TEXT NOT NULL,
    created_at TEXT NOT NULL
);

-- =========================================================
-- TABLE: runs
-- Logged system queries and outputs (audit trail)
-- =========================================================
CREATE TABLE runs (
    run_id TEXT PRIMARY KEY,
    input_query TEXT NOT NULL,
    output_text TEXT NOT NULL,
    cited_chunk_ids TEXT,                -- JSON array stored as TEXT
    created_at TEXT NOT NULL,
    verdict TEXT NOT NULL                -- ok | unknown | flagged
);

-- =========================================================
-- END OF SCHEMA

def utc_now_iso() -> str:

def append_jsonl(path: Path, obj: Dict[str, Any]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("a", encoding="utf-8") as f:

# scripts/hash_utils.py
from __future__ import annotations

import hashlib
from pathlib import Path
from typing import Dict, List, Tuple


def sha256_file(path: Path, chunk_size: int = 1024 * 1024) -> str:
    h = hashlib.sha256()
    with path.open("rb") as f:
        for chunk in iter(lambda: f.read(chunk_size), b""):
            h.update(chunk)
    return h.hexdigest()


def list_files(root: Path) -> List[Path]:
    # Deterministic ordering
    return sorted([p for p in root.rglob("*") if p.is_file()])


def sha256_manifest(root: Path) -> List[Tuple[str, str]]:
    """
    Returns list of (relative_path, sha256) for every file under root, sorted.
    """
    files = list_files(root)
    out: List[Tuple[str, str]] = []
    for f in files:
        rel = f.relative_to(root).as_posix()
        out.append((rel, sha256_file(f)))
    return out

# scripts/run_recorded.py
from __future__ import annotations

import subprocess
import sys
from pathlib import Path

from hash_utils import sha256_file
from ops_log import log_event

BASE_DIR = Path(__file__).resolve().parent.parent
DATA_DIR = BASE_DIR / "data"
SCHEMA_PATH = DATA_DIR / "schema.sql"
DB_PATH = DATA_DIR / "memory.db"
LOGS_DIR = BASE_DIR / "logs"


def main() -> int:
    if len(sys.argv) < 2:
        print("Usage: python scripts/run_recorded.py <script_path> [args...]")
        return 2

    script_path = Path(sys.argv[1]).resolve()
    args = sys.argv[2:]

    if not script_path.exists():
        print(f"Script not found: {script_path}")
        return 2

    LOGS_DIR.mkdir(parents=True, exist_ok=True)

    schema_sha = sha256_file(SCHEMA_PATH) if SCHEMA_PATH.exists() else None
    db_sha_before = sha256_file(DB_PATH) if DB_PATH.exists() else None

    cmd = [sys.executable, str(script_path)] + args
    proc = subprocess.run(cmd, capture_output=True, text=True)

    # Store stdout/stderr to files for audit
    run_tag = f"{script_path.stem}"
    out_path = LOGS_DIR / f"{run_tag}.stdout.log"
    err_path = LOGS_DIR / f"{run_tag}.stderr.log"
    out_path.write_text(proc.stdout, encoding="utf-8")
    err_path.write_text(proc.stderr, encoding="utf-8")

    db_sha_after = sha256_file(DB_PATH) if DB_PATH.exists() else None

    log_event(
        action="run_script",
        payload={"cmd": cmd},
        artifacts={
            "exit_code": proc.returncode,
            "stdout_path": str(out_path.as_posix()),
            "stderr_path": str(err_path.as_posix()),
            "schema_sha256": schema_sha,
            "db_sha256_before": db_sha_before,
            "db_sha256_after": db_sha_after,
        },
    )

    print(proc.stdout, end="")
    if proc.stderr:
        print(proc.stderr, file=sys.stderr, end="")

    return proc.returncode


if __name__ == "__main__":
    raise SystemExit(main())

from __future__ import annotations

import json
from pathlib import Path

State = Literal["OBSERVE", "RECORD", "EXECUTE"]

# What is allowed in each state (keep minimal + explicit)
ALLOWED_ACTIONS = {
    "OBSERVE": set(),  # do nothing
    "RECORD": {"snapshot_anchors", "run_script"},
    "EXECUTE": {"snapshot_anchors", "run_script"},
}

def get_state() -> State:
    if not STATE_PATH.exists():
        # STGRAIL rule: If state is undeclared -> do nothing
        raise RuntimeError(f"STATE is undeclared. Missing: {STATE_PATH}")

    data = json.loads(STATE_PATH.read_text(encoding="utf-8"))
    state = data.get("state")
    if state not in ("OBSERVE", "RECORD", "EXECUTE"):
        raise RuntimeError(f"Invalid state in STATE.json: {state}")
    return state  # type: ignore[return-value]

def require_allowed(action: str) -> None:
    state = get_state()
    allowed = ALLOWED_ACTIONS.get(state, set())

# scripts/init_db.py
from __future__ import annotations

import os
import sqlite3
import hashlib
from pathlib import Path


def sha256_file(path: Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()


def require_recorded_execution() -> None:
    # Hard gate: init_db must only run via run_recorded.py
    if os.environ.get("SOLOB_RECORDED_RUN") != "1":
        raise RuntimeError(
            "Blocked: init_db.py refuses unrecorded execution.\n"
            "Run via: python scripts/run_recorded.py --intent \"...\" scripts/init_db.py"
        )
    # Optional but strong: require explicit human intent to be passed through env
    intent = os.environ.get("SOLOB_RUN_INTENT", "").strip()
    if not intent:
        raise RuntimeError(
            "Blocked: Missing SOLOB_RUN_INTENT. "
            "run_recorded.py must pass human intent into the environment."
        )


def init_db() -> None:
    require_recorded_execution()

    base_dir = Path(__file__).resolve().parent.parent
    data_dir = base_dir / "data"
    schema_path = data_dir / "schema.sql"
    db_path = data_dir / "memory.db"

    if not schema_path.exists():
        raise FileNotFoundError(f"schema.sql not found: {schema_path}")

    # Strict: refuse to touch an existing DB file (prevents accidental overwrites / ambiguity)
    if db_path.exists():
        raise RuntimeError(
            f"Blocked: {db_path} already exists.\n"
            "Policy: init_db is one-time creation only.\n"
            "If you intend a fresh initialization, quarantine/remove the existing DB as a recorded action."
        )

    data_dir.mkdir(parents=True, exist_ok=True)

    schema_sql = schema_path.read_text(encoding="utf-8")
    schema_sha = sha256_file(schema_path)

    # Create DB + apply schema verbatim in one transaction
    conn = sqlite3.connect(db_path)
    try:
        conn.execute("PRAGMA foreign_keys = ON;")
        conn.executescript(schema_sql)
        conn.commit()
    except Exception:
        conn.rollback()
        # If schema application fails, remove the newly-created DB file to avoid half-built artifacts
        try:
            conn.close()
        finally:
            if db_path.exists():
                db_path.unlink(missing_ok=True)
        raise
    finally:
        try:
            conn.close()
        except Exception:
            pass

    print("[OK] init_db complete (recorded execution required).")
    print(f"DB: {db_path}")
    print(f"Schema SHA256: {schema_sha}")


if __name__ == "__main__":
    init_db()
